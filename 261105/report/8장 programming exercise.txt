1번. FindLargest Procedure (부호 있는 DWORD 배열의 최댓값 찾기)
[문제]
FindLargest가 signed DWORD 배열의 주소와 원소 개수를 받아 EAX에 최댓값을 반환하게 한다.
PROC 매개변수와 PROTO를 사용하고, EAX를 제외한 변경 레지스터를 보존한다.
길이가 다른 배열 3개를 시험하며, 음수도 포함한다.

[정답 - 전체 코드]
INCLUDE Irvine32.inc
FindLargest PROTO, pArray:PTR SDWORD, n:DWORD

.data
arrOne   SDWORD -7,-2,-30,-11
arrTwo   SDWORD 10,95,-42,7,81,3
arrThree SDWORD -100,-50,-90
msg1 BYTE "Array 1 max = ",0
msg2 BYTE "Array 2 max = ",0
msg3 BYTE "Array 3 max = ",0

.code
main PROC
    mov edx,OFFSET msg1
    call WriteString
    INVOKE FindLargest,ADDR arrOne,LENGTHOF arrOne
    call WriteInt
    call Crlf

    mov edx,OFFSET msg2
    call WriteString
    INVOKE FindLargest,ADDR arrTwo,LENGTHOF arrTwo
    call WriteInt
    call Crlf

    mov edx,OFFSET msg3
    call WriteString
    INVOKE FindLargest,ADDR arrThree,LENGTHOF arrThree
    call WriteInt
    call Crlf
    exit
main ENDP

FindLargest PROC USES esi ecx, pArray:PTR SDWORD, n:DWORD
    mov esi,pArray
    mov ecx,n
    test ecx,ecx
    jz EmptyArray             ; 입력은 원소가 1개 이상이어야 정상적인 최댓값 정의
    mov eax,[esi]            ; 첫 원소를 기준값으로
    dec ecx
    jz Finished
L1:
    add esi,TYPE SDWORD
    cmp DWORD PTR [esi],eax
    jle Next                 ; signed 비교
    mov eax,[esi]
Next:
    loop L1
Finished:
    ret
EmptyArray:
    xor eax,eax             ; 예제의 방어 처리. 빈 배열의 최댓값은 정의되지 않음
    ret
FindLargest ENDP
END main

[예상 결과]
Array 1 max = -2
Array 2 max = +95 (WriteInt 표시에 따라 양수 기호가 출력될 수 있음)
Array 3 max = -50

[풀이]
ESI가 배열을 순서대로 읽고 ECX가 반복 횟수를 담당한다.
첫 원소를 EAX에 넣어 기준으로 사용한 뒤, JLE를 통해 큰 값을 발견했을 때만 EAX를 바꾼다.
음수 배열도 처리해야 하므로 JA/JB가 아닌 signed 조건 점프를 사용한다.

2번. Chess Board (회색/흰색 8×8 체스판)
[문제]
회색과 흰색이 교대로 배치된 8×8 체스판을 그린다.
SetTextColor와 Gotoxy를 사용하고, 전역변수 대신 프로시저와 매개변수를 사용한다.

[정답 - 전체 코드]
INCLUDE Irvine32.inc

.code
main PROC
    call DrawBoard
    call WaitMsg
    exit
main ENDP

; 색상과 커서 이동으로 8×8 배경 사각형(각 칸 2칸 너비)을 표현한다.
DrawBoard PROC USES ebx esi edx
    xor ebx,ebx            ; EBX = row(0~7)
RowLoop:
    mov edx,0
    mov dh,bl              ; DH = row
    mov dl,0               ; DL = column
    call Gotoxy
    xor esi,esi            ; ESI = col(0~7)
ColLoop:
    mov eax,ebx
    add eax,esi
    and eax,1
    jz GraySquare
    mov eax,black + (white * 16)
    jmp SetSquareColor
GraySquare:
    mov eax,black + (lightGray * 16)
SetSquareColor:
    call SetTextColor
    mov al,' '
    call WriteChar
    mov al,' '
    call WriteChar
    inc esi
    cmp esi,8
    jb ColLoop
    inc ebx
    cmp ebx,8
    jb RowLoop

    mov eax,lightGray + (black * 16)  ; 색 복원
    call SetTextColor
    mov edx,0
    mov dh,8
    call Gotoxy
    call Crlf
    ret
DrawBoard ENDP
END main

[예상 결과]
8행×8열로 회색/흰색 배경 칸이 교대로 표시된다.
콘솔 테마, 글꼴 및 콘솔 창 크기에 따라 실제 칸의 화면 비율은 다를 수 있다.

[풀이]
행과 열을 더한 값이 짝수인지 홀수인지로 색을 바꾸면 체스판 무늬가 만들어진다.
Gotoxy의 DH는 행, DL은 열이고, SetTextColor는 EAX에 색 속성을 받는다.

3번. Chess Board with Alternating Colors (16가지 배경색)
[문제]
2번 체스판을 확장한다. 회색 칸만 500ms마다 색을 바꾸어 모든 4비트 배경색
16가지를 차례로 표시한다. 흰 칸은 계속 흰색으로 둔다.

[정답 - 전체 코드]
INCLUDE Irvine32.inc
DrawBoard PROTO, changeColor:DWORD

.code
main PROC
    xor esi,esi            ; ESI = 0~15 배경색 인덱스
L1:
    INVOKE DrawBoard,esi
    mov eax,500            ; 500ms
    call Delay
    inc esi
    cmp esi,16
    jb L1
    mov eax,lightGray + (black * 16)
    call SetTextColor
    call Crlf
    exit
main ENDP

DrawBoard PROC USES ebx edi edx, changeColor:DWORD
    xor ebx,ebx            ; row
RowLoop:
    mov edx,0
    mov dh,bl
    mov dl,0
    call Gotoxy
    xor edi,edi            ; col
ColLoop:
    mov eax,ebx
    add eax,edi
    and eax,1
    jz ChangingSquare
    mov eax,black + (white * 16)      ; 흰 칸: 항상 흰색
    jmp DrawSquare
ChangingSquare:
    mov eax,changeColor
    shl eax,4                         ; 배경색 * 16
    or eax,black                      ; 검은색 전경
DrawSquare:
    call SetTextColor
    mov al,' '
    call WriteChar
    mov al,' '
    call WriteChar
    inc edi
    cmp edi,8
    jb ColLoop
    inc ebx
    cmp ebx,8
    jb RowLoop
    ret
DrawBoard ENDP
END main

[예상 결과]
약 0.5초 간격으로 색이 바뀌는 체스판 16회 표시 (총 지연 8초).
변화 칸의 배경색 인덱스는 0,1,2,...,15이고 흰 칸은 고정이다.

[풀이]
색상 숫자를 매개변수로 전달하고 프로시저를 반복 호출한다.
배경색은 상위 4비트이므로 SHL 4로 이동한 뒤 전경색과 합친다.
0~15 중 15에서는 변화 칸도 흰색이므로 두 종류의 칸이 같은 색으로 보일 수 있다.

4번. FindThrees Procedure (연속된 3 세 개 찾기)
[문제]
배열에서 3,3,3이 연속으로 등장하는 구간이 있으면 EAX=1, 없으면 EAX=0을 반환한다.
배열 주소와 길이를 매개변수로 받아서 여러 배열로 테스트한다.

[정답 - 전체 코드]
INCLUDE Irvine32.inc
FindThrees PROTO, pArray:PTR SDWORD, n:DWORD

.data
arr1 SDWORD 1,3,3,3,9
arr2 SDWORD 3,3,2,3,3
arr3 SDWORD 3,3,3
arr4 SDWORD 1,2
msg BYTE "FindThrees result = ",0

.code
main PROC
    mov edx,OFFSET msg
    call WriteString
    INVOKE FindThrees,ADDR arr1,LENGTHOF arr1
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE FindThrees,ADDR arr2,LENGTHOF arr2
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE FindThrees,ADDR arr3,LENGTHOF arr3
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE FindThrees,ADDR arr4,LENGTHOF arr4
    call WriteDec
    call Crlf
    exit
main ENDP

FindThrees PROC USES esi ecx, pArray:PTR SDWORD, n:DWORD
    mov esi,pArray
    mov ecx,n
    cmp ecx,3
    jb NotFound
    sub ecx,2             ; 길이 n이면 시작 가능한 인덱스는 n-2개
L1:
    cmp DWORD PTR [esi],3
    jne Next
    cmp DWORD PTR [esi+4],3
    jne Next
    cmp DWORD PTR [esi+8],3
    jne Next
    mov eax,1
    ret
Next:
    add esi,TYPE SDWORD
    loop L1
NotFound:
    xor eax,eax
    ret
FindThrees ENDP
END main

[예상 결과]
1 / 0 / 1 / 0

[풀이]
3개 묶음을 읽으므로 검사 시작 위치는 0부터 n-3까지다.
DWORD 3개는 [esi], [esi+4], [esi+8]이므로 세 위치 모두 3인지 확인한다.

5번. DifferentInputs Procedure (세 값이 모두 서로 다른지 확인)
[문제]
세 입력값이 모두 서로 다르면 EAX=1, 하나라도 같으면 EAX=0을 반환한다.
PROC 매개변수와 PROTO를 사용하고 다섯 번 호출하여 시험한다.

[정답 - 전체 코드]
INCLUDE Irvine32.inc
DifferentInputs PROTO, a:DWORD, b:DWORD, c:DWORD

.data
msg BYTE "Different = ",0

.code
main PROC
    mov edx,OFFSET msg
    call WriteString
    INVOKE DifferentInputs,1,2,3
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE DifferentInputs,1,1,3
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE DifferentInputs,1,2,1
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE DifferentInputs,2,1,1
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE DifferentInputs,-1,0,2
    call WriteDec
    call Crlf
    exit
main ENDP

DifferentInputs PROC USES edx, a:DWORD, b:DWORD, c:DWORD
    xor eax,eax
    mov edx,a
    cmp edx,b
    je Finished
    cmp edx,c
    je Finished
    mov edx,b
    cmp edx,c
    je Finished
    mov eax,1
Finished:
    ret
DifferentInputs ENDP
END main

[예상 결과]
1 / 0 / 0 / 0 / 1

[풀이]
a!=b, a!=c, b!=c 세 비교가 모두 참일 때만 1을 반환한다.

6번. Exchanging Integers (임의 정수 배열에서 인접한 두 원소 교환)
[문제]
임의의 순서로 된 정수 배열을 만들고 8.4.6절의 Swap 프로시저를 사용하여
반복문에서 연속된 원소 쌍을 교환하라.

[정답 - 전체 코드]
INCLUDE Irvine32.inc
Swap PROTO, pLeft:PTR DWORD, pRight:PTR DWORD

.data
nums DWORD 10 DUP(?)           ; 원소 10개 = 5쌍
msg1 BYTE "Before swap:",0
msg2 BYTE "After swap:",0

.code
main PROC
    call Randomize
    xor esi,esi
FillArray:
    mov eax,100                ; 0~99의 임의 값
    call RandomRange
    mov nums[esi],eax
    add esi,TYPE nums
    cmp esi,SIZEOF nums
    jb FillArray

    mov edx,OFFSET msg1
    call WriteString
    call Crlf
    mov esi,OFFSET nums
    mov ecx,LENGTHOF nums
    mov ebx,TYPE nums
    call DumpMem

    xor esi,esi                ; (0,1), (2,3), ..., (8,9) 교환
SwapLoop:
    lea eax,nums[esi]
    lea edx,nums[esi+4]
    INVOKE Swap,eax,edx       ; ADDR에 변하는 인덱스를 넣지 않고 LEA 활용
    add esi,8                 ; 다음 2개 DWORD 쌍
    cmp esi,SIZEOF nums
    jb SwapLoop

    mov edx,OFFSET msg2
    call WriteString
    call Crlf
    mov esi,OFFSET nums
    mov ecx,LENGTHOF nums
    mov ebx,TYPE nums
    call DumpMem
    exit
main ENDP

Swap PROC USES eax esi edi, pLeft:PTR DWORD, pRight:PTR DWORD
    mov esi,pLeft
    mov edi,pRight
    mov eax,[esi]
    xchg eax,[edi]
    mov [esi],eax
    ret
Swap ENDP
END main

[예상 결과]
난수 배열의 값은 실행마다 달라질 수 있다.
교환 전 원소가 [a,b,c,d,e,f,...]라면 교환 후 [b,a,d,c,f,e,...]가 된다.

[풀이]
Swap이 두 DWORD 값의 주소를 받아 XCHG로 바꾼다.
ESI는 8바이트씩 이동해야 다음 인접한 두 원소를 가리킨다.
주소가 실행 중 바뀌므로 LEA로 실제 포인터를 구해 INVOKE에 전달한다.

7번. Greatest Common Divisor (재귀 유클리드 최대공약수)
[문제]
GCD(x,y)를 재귀로 구현하고 다음 입력을 시험한다.
(5,20), (24,18), (11,7), (432,226), (26,13)

[정답 - 전체 코드]
INCLUDE Irvine32.inc
GCD PROTO, x:DWORD, y:DWORD

.data
msg BYTE "GCD = ",0

.code
main PROC
    mov edx,OFFSET msg
    call WriteString
    INVOKE GCD,5,20
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE GCD,24,18
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE GCD,11,7
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE GCD,432,226
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE GCD,26,13
    call WriteDec
    call Crlf
    exit
main ENDP

GCD PROC USES ebx edx, x:DWORD, y:DWORD
    mov eax,x
    test eax,eax
    jns XNonnegative
    neg eax                ; 절댓값의 unsigned 비트 패턴
XNonnegative:
    mov ebx,y
    test ebx,ebx
    jns YNonnegative
    neg ebx
YNonnegative:
    test ebx,ebx
    jz Done                ; 종료: y=0이면 gcd(x,0)=|x|
    xor edx,edx
    div ebx                ; EDX = x % y (unsigned 나머지)
    INVOKE GCD,ebx,edx    ; 재귀: gcd(y, x % y)
Done:
    ret
GCD ENDP
END main

[예상 결과]
5 / 6 / 1 / 2 / 13

[풀이]
유클리드 알고리즘: gcd(x,y) = gcd(y, x mod y).
y가 0이 되면 재귀 호출을 멈추고 x를 반환한다.
매 단계의 복귀 주소와 매개변수는 스택에 따로 저장된다.
입력이 모두 0이면 수학적으로 최대공약수를 정의할 수 없지만, 이 예시는 0을 반환한다.

8번. Counting Matching Elements (동일 위치 값이 같은 원소 수)
[문제]
두 signed DWORD 배열의 주소와 길이를 받는 CountMatches를 작성한다.
같은 위치의 값이 같으면 1씩 증가시키고 총 일치 개수를 EAX에 반환한다.
PROTO/INVOKE를 사용하며 EAX 이외의 변경 레지스터는 보존한다.

[정답 - 전체 코드]
INCLUDE Irvine32.inc
CountMatches PROTO, pA:PTR SDWORD, pB:PTR SDWORD, n:DWORD

.data
arrA SDWORD 10,20,-3,40,50
arrB SDWORD 10,0,-3,80,50
arrC SDWORD -1,-2,7,7
arrD SDWORD -1,-2,7,0
msg BYTE "Matching count = ",0

.code
main PROC
    mov edx,OFFSET msg
    call WriteString
    INVOKE CountMatches,ADDR arrA,ADDR arrB,LENGTHOF arrA
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE CountMatches,ADDR arrC,ADDR arrD,LENGTHOF arrC
    call WriteDec
    call Crlf
    exit
main ENDP

CountMatches PROC USES esi edi ecx edx,
    pA:PTR SDWORD, pB:PTR SDWORD, n:DWORD
    mov esi,pA
    mov edi,pB
    mov ecx,n
    xor eax,eax
    jecxz Done
L1:
    mov edx,[esi]
    cmp edx,[edi]
    jne Skip
    inc eax
Skip:
    add esi,TYPE SDWORD
    add edi,TYPE SDWORD
    loop L1
Done:
    ret
CountMatches ENDP
END main

[예상 결과]
3 / 3

[풀이]
ESI와 EDI를 각각 4바이트씩 이동하며 대응하는 두 원소를 비교한다.
서로 같은 위치에서 값이 같을 때만 EAX를 1 증가시킨다.

9번. Counting Nearly Matching Elements (값 차이가 제한 이하인 원소 수)
[문제]
두 signed DWORD 배열과 원소 개수, 최대 허용 차이(diff)를 받는다.
각 위치의 두 값 차이의 절댓값이 diff 이하이면 1씩 세어 EAX로 반환한다.
INVOKE/PROTO를 사용하고 EAX 외에 변경 레지스터는 보존한다.

[정답 - 전체 코드]
INCLUDE Irvine32.inc
CountNearMatches PROTO, pA:PTR SDWORD, pB:PTR SDWORD, n:DWORD, maxDiff:DWORD

.data
arrA SDWORD 1,10,-20,40,100
arrB SDWORD 3,14,-25,48,95
arrC SDWORD 80000000h,7,-100,50       ; INT32_MIN, 7, -100, 50
arrD SDWORD 7FFFFFFFh,6,-101,51       ; INT32_MAX, 6, -101, 51
msg BYTE "Nearly matching count = ",0

.code
main PROC
    mov edx,OFFSET msg
    call WriteString
    INVOKE CountNearMatches,ADDR arrA,ADDR arrB,LENGTHOF arrA,5
    call WriteDec
    call Crlf

    mov edx,OFFSET msg
    call WriteString
    INVOKE CountNearMatches,ADDR arrC,ADDR arrD,LENGTHOF arrC,2
    call WriteDec
    call Crlf
    exit
main ENDP

CountNearMatches PROC USES ebx ecx edx esi edi,
    pA:PTR SDWORD, pB:PTR SDWORD, n:DWORD, maxDiff:DWORD
    mov esi,pA
    mov edi,pB
    mov ecx,n
    xor ebx,ebx               ; EBX = 일치 개수
    jecxz Done
L1:
    mov eax,[esi]             ; x
    mov edx,[edi]             ; y
    cmp eax,edx
    jge XIsLarger             ; signed 비교로 큰 값 결정
    sub edx,eax               ; y - x (unsigned 32비트 차이)
    mov eax,edx
    jmp CheckDifference
XIsLarger:
    sub eax,edx               ; x - y (unsigned 32비트 차이)
CheckDifference:
    cmp eax,maxDiff           ; 차이와 제한은 unsigned 비교
    ja Next
    inc ebx
Next:
    add esi,TYPE SDWORD
    add edi,TYPE SDWORD
    loop L1
Done:
    mov eax,ebx
    ret
CountNearMatches ENDP
END main

[예상 결과]
4 / 3

[풀이]
첫 번째 테스트의 차이는 2,4,5,8,5이므로 diff=5 이하가 4개다.
두 번째 테스트는 INT32_MIN과 INT32_MAX가 포함된다. 이 차이는 4294967295로
signed 32비트에는 못 담지만 unsigned 32비트에는 담을 수 있으므로
먼저 signed 비교로 큰 값을 판단하고, 뺄셈 결과를 unsigned로 비교한다.
maxDiff는 0 이상인 DWORD로 전달한다고 가정한다.

10번. Show Procedure Parameters (스택 매개변수의 주소와 16진수 값 출력)
[문제]
ShowParams가 호출자 프로시저의 32비트 스택 매개변수들을
낮은 주소에서 높은 주소 순서로, 각 주소와 값을 16진수로 출력하도록 한다.
표시할 매개변수 개수 하나를 ShowParams에 전달한다.
예: INVOKE MySample,1234h,5000h,6543h

[정답 - 전체 코드]
INCLUDE Irvine32.inc
ShowParams PROTO, n:DWORD
MySample PROTO, first:DWORD, second:DWORD, third:DWORD

.data
header BYTE "Stack parameters:",0
addrMsg BYTE "Address ",0
equalMsg BYTE " = ",0

.code
main PROC
    INVOKE MySample,1234h,5000h,6543h
    exit
main ENDP

MySample PROC, first:DWORD, second:DWORD, third:DWORD
    INVOKE ShowParams,3
    ret
MySample ENDP

ShowParams PROC USES ebx esi edx, n:DWORD
    mov edx,OFFSET header
    call WriteString
    call Crlf

    mov esi,[ebp]            ; ShowParams에 저장된 호출자(MySample)의 EBP
    add esi,8                ; MySample의 첫 매개변수 주소
    mov ebx,n
    cmp ebx,0
    je Finished
L1:
    mov edx,OFFSET addrMsg
    call WriteString
    mov eax,esi
    call WriteHex            ; 스택 매개변수 자체의 주소
    mov edx,OFFSET equalMsg
    call WriteString
    mov eax,[esi]
    call WriteHex            ; 해당 주소에 저장된 32비트 값
    call Crlf
    add esi,4               ; 다음 DWORD 매개변수
    dec ebx
    jnz L1
Finished:
    ret
ShowParams ENDP
END main

[예상 출력 형식]
Stack parameters:
Address XXXXXXXX = 00001234
Address XXXXXXXX = 00005000
Address XXXXXXXX = 00006543
(XXXXXXXX은 실행 환경에서 결정되는 실제 스택 주소)

[풀이]
MySample은 확장 PROC 선언으로 EBP 스택 프레임을 가진다.
ShowParams의 [EBP]에는 호출자인 MySample의 EBP 값이 저장된다.
MySample에서 [EBP+8]이 첫 인수이고 나머지는 4바이트 간격으로 이어진다.
그러므로 ShowParams에서 ESI를 호출자 EBP+8로 설정한 뒤 4씩 증가시킨다.
실제 주소는 실행할 때마다 달라질 수 있으므로 교재 예시 주소를 고정 출력하면 안 된다.
