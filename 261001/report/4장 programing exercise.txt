 
1. Converting from Big Endian to Little Endian
 

[문제 해석]
bigEndian의 바이트 순서를 반대로 뒤집어 littleEndian에 복사하는 프로그램을 작성하라.
bigEndian의 32비트 값은 12345678h로 간주한다.

.data
bigEndian    BYTE 12h,34h,56h,78h
littleEndian DWORD ?

[답]

mov al,bigEndian
mov BYTE PTR littleEndian+3,al

mov al,bigEndian+1
mov BYTE PTR littleEndian+2,al

mov al,bigEndian+2
mov BYTE PTR littleEndian+1,al

mov al,bigEndian+3
mov BYTE PTR littleEndian,al


 
2. Exchanging Pairs of Array Values
 

[문제 해석]
짝수 개의 원소를 가진 배열에서 모든 인접한 두 값을 서로 교환하는 프로그램을 작성하라.
즉 i번째 원소는 i+1번째 원소와, i+2번째 원소는 i+3번째 원소와 교환한다.

[답]

.data
array DWORD 1,2,3,4,5,6

.code
mov esi,0
mov ecx,LENGTHOF array / 2

L1:
    mov eax,array[esi]
    mov ebx,array[esi+4]

    mov array[esi],ebx
    mov array[esi+4],eax

    add esi,8
    loop L1


 
3. Summing the Gaps between Array Values
 

[문제 해석]
오름차순으로 정렬된 DWORD 배열에서 연속된 원소들 사이의 차이값을 모두 더하는 프로그램을 작성하라.
예: {0,2,5,9,10}이면 차이는 2,3,4,1이고 합은 10이다.

[답]

.data
array DWORD 0,2,5,9,10
sum   DWORD ?

.code
mov eax,0
mov esi,0
mov ecx,LENGTHOF array - 1

L1:
    mov ebx,array[esi+4]
    sub ebx,array[esi]
    add eax,ebx

    add esi,4
    loop L1

mov sum,eax


 
4. Copying a Word Array to a DoubleWord Array
 

[문제 해석]
unsigned WORD(16비트) 배열의 모든 원소를 unsigned DWORD(32비트) 배열로 복사하는 프로그램을 작성하라.

[답]

.data
wordArray  WORD 10h,20h,30h,40h
dwordArray DWORD 4 DUP(?)

.code
mov esi,0
mov edi,0
mov ecx,LENGTHOF wordArray

L1:
    movzx eax,wordArray[esi]
    mov dwordArray[edi],eax

    add esi,2
    add edi,4
    loop L1


 
5. Fibonacci Numbers
 

[문제 해석]
반복문을 사용하여 피보나치 수열의 처음 7개 값을 계산하는 프로그램을 작성하라.
Fib(1)=1, Fib(2)=1, Fib(n)=Fib(n-1)+Fib(n-2)

[답]

.data
fib DWORD 7 DUP(?)

.code
mov DWORD PTR fib[0],1
mov DWORD PTR fib[4],1

mov esi,8
mov ecx,5

L1:
    mov eax,fib[esi-8]
    add eax,fib[esi-4]
    mov fib[esi],eax

    add esi,4
    loop L1


 
6. Reverse an Array
 

[문제 해석]
다른 배열에 복사하지 않고, 현재 배열 안에서 원소 순서를 반대로 뒤집는 프로그램을 작성하라.
SIZEOF, TYPE, LENGTHOF를 사용하라.

[답]

.data
array DWORD 1,2,3,4,5

.code
mov esi,0
mov edi,SIZEOF array - TYPE array
mov ecx,LENGTHOF array / 2

L1:
    mov eax,array[esi]
    xchg eax,array[edi]
    mov array[esi],eax

    add esi,TYPE array
    sub edi,TYPE array
    loop L1


 
7. Copy a String in Reverse Order
 

[문제 해석]
source 문자열을 target으로 복사하되 문자 순서를 반대로 뒤집어 복사하는 프로그램을 작성하라.

source BYTE "This is the source string",0
target BYTE SIZEOF source DUP('#')

[답]

mov esi,OFFSET source
add esi,SIZEOF source - 2

mov edi,OFFSET target
mov ecx,SIZEOF source - 1

L1:
    mov al,[esi]
    mov [edi],al

    dec esi
    inc edi
    loop L1

mov BYTE PTR [edi],0


 
8. Shifting the Elements in an Array
 

[문제 해석]
32비트 정수 배열의 모든 원소를 한 칸 앞으로 회전시키는 프로그램을 작성하라.
마지막 원소는 첫 번째 위치로 이동해야 한다.
예: {10,20,30,40} -> {40,10,20,30}

[답]

.data
array DWORD 10,20,30,40

.code
mov eax,array[SIZEOF array-TYPE array]

mov esi,SIZEOF array-TYPE array
mov ecx,LENGTHOF array-1

L1:
    mov ebx,array[esi-4]
    mov array[esi],ebx

    sub esi,4
    loop L1

mov array[0],eax
